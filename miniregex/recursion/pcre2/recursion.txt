# The nested notes: (?R) calls the whole pattern
# 入れ子の注記：(?R) はパターン全体を呼ぶ
/\((?:[^()]++|(?R))*\)/g
    Mira (healer (2nd class) from Lam (east)) paid 120 gold (cash)

# A call does not set the group it calls
# 呼び出しは、呼んだグループを書き換えない
/(sens|respons)e and (?1)ibility/
    sense and responsibility

/(?(DEFINE)(?<d>a))(?&d)\k<d>/
    aa

# Back into a recursion: the palindromes
# 再帰の中へ戻る：回文
/^((.)(?1)\2|.?)$/
    abcba
    abba
    abca

# Going round at one place
# 同じ場所で回る
/a|(?R)b/
    xb

# Without ++, the steps double for each a
# ++ がないと、a が1つ増えるごとにステップ数が2倍
/\((?:[^()]+|(?R))*\)/
    (aaaaaaaaaaaaaaaaaaaaaaaaaaaaaa()

/\((?:[^()]++|(?R))*\)/
    (aaaaaaaaaaaaaaaaaaaaaaaaaaaaaa()

# The same place, but a look makes group 2 grow: the calls end
# 同じ場所だが、「見る」ものがグループ2を伸ばす：呼び出しは終わる
/^(?<r>(?=(\2a|a))(?:\2$|(?&r)))$/
    aaa

# PCRE2 stops when the place and how far it has read are the same
# PCRE2 は「場所と、どこまで読んだか」が同じなら止める
/(?1)(?!)|(|(?>(?1))y)/no_start_optimize,no_auto_possess
    a

/(?1)x|(|(?1)y)/no_start_optimize,no_auto_possess
    a

# x has read the a: PCRE2 goes on (MiniRegex stops: Looping 1 0)
# x が a を読んだ：PCRE2 は進む（MiniRegex は止める：Looping 1 0）
/(?1)x|(|(?>(?1))y)/no_start_optimize,no_auto_possess
    a

# .* has read to the end first: PCRE2 stops (MiniRegex finds aaa)
# 先に .* が終わりまで読んだ：PCRE2 は止める（MiniRegex は aaa を見つける）
/^(?<r>(?=.*)(?=(\2a|a))(?:\2$|(?&r)))$/
    aaa
    aaa\=disable_recurseloop_check

# \K in a called group counts
# 呼んだグループの中の \K も効く
/(a\Kb)?c(?1)/
    cab

# A DEFINE group has one branch only
# DEFINE のグループの枝は1本だけ
/(?(DEFINE)a|b)/

# The answer of step 1
# ステップ1の解答
/(a|(?1)b)/
    b
